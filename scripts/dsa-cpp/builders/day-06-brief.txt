std::vector<int> stable_ids(const std::vector<int> &scores, const std::vector<int> &ids) {
    int n = (int)scores.size();
    std::vector<int> order(n);
    for (int i = 0; i < n; i++) order[i] = i;
    for (int i = 1; i < n; i++) {
        int cur = order[i], j = i - 1;
        while (j >= 0 && scores[order[j]] > scores[cur]) {
            order[j + 1] = order[j];
            j--;
        }
        order[j + 1] = cur;
    }
    std::vector<int> out(n);
    for (int i = 0; i < n; i++) out[i] = ids[order[i]];
    return out;
}

std::vector<int> rank_products(const std::vector<int> &prices, const std::vector<int> &ids) {
    int n = (int)prices.size();
    std::vector<int> order(n);
    for (int i = 0; i < n; i++) order[i] = i;
    for (int i = 0; i < n - 1; i++) {
        int m = i;
        for (int j = i + 1; j < n; j++) {
            bool better = prices[order[j]] > prices[order[m]] ||
                (prices[order[j]] == prices[order[m]] &&
                 ids[order[j]] < ids[order[m]]);
            if (better) m = j;
        }
        if (m != i) {
            int t = order[i];
            order[i] = order[m];
            order[m] = t;
        }
    }
    std::vector<int> out(n);
    for (int i = 0; i < n; i++) out[i] = ids[order[i]];
    return out;
}

std::vector<int> freq_order(std::vector<int> v) {
    int n = (int)v.size();
    for (int i = 0; i < n - 1; i++) {
        int m = i, mf = -1;
        for (int j = i; j < n; j++) {
            int f = 0;
            for (int k = 0; k < n; k++)
                if (v[k] == v[j]) f++;
            bool better = f > mf || (f == mf && v[j] < v[m]);

                m = j;
                mf = f;
            }
        }
        if (m != i) {
            int t = v[i];
            v[i] = v[m];
            v[m] = t;
        }
    }
    return v;
}

std::vector<int> selection_after_k(std::vector<int> v, int k) {
    int n = (int)v.size();
    for (int i = 0; i < k && i < n - 1; i++) {
        int m = i;
        for (int j = i + 1; j < n; j++)
            if (v[j] < v[m]) m = j;
        if (m != i) {
            int t = v[i];
            v[i] = v[m];
            v[m] = t;
        }
    }
    return v;
}

std::vector<int> insertion_after_k(std::vector<int> v, int k) {
    int n = (int)v.size();
    for (int i = 1; i <= k && i < n; i++) {
        int key = v[i], j = i - 1;
        while (j >= 0 && v[j] > key) {
            v[j + 1] = v[j];
            j--;
        }
        v[j + 1] = key;
    }
    return v;
}

std::vector<std::string> sort_words(std::vector<std::string> words) {
    int n = (int)words.size();
    for (int i = 1; i < n; i++) {
        std::string key = words[i];
        int j = i - 1;
        while (j >= 0 && (words[j].size() > key.size() ||
                          (words[j].size() == key.size() && words[j] > key))) {
            words[j + 1] = words[j];
            j--;
        }
        words[j + 1] = key;
    }
    return words;
}

std::vector<std::vector<int>> bubble_trace(std::vector<int> v) {
    int n = (int)v.size();
    std::vector<std::vector<int>> rows;
    bool swapped = true;
    while (swapped) {
        swapped = false;
