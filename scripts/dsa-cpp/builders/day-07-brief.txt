void merge_ranges(std::vector<int> &v, int lo, int mid, int hi) {
    std::vector<int> t;
    int i = lo, j = mid;
    while (i < mid && j < hi)
        t.push_back(v[i] <= v[j] ? v[i++] : v[j++]);
    while (i < mid) t.push_back(v[i++]);
    while (j < hi) t.push_back(v[j++]);
    for (int k = 0; k < (int)t.size(); k++) v[lo + k] = t[k];
}

void msort_rec(std::vector<int> &v, int lo, int hi) {
    if (hi - lo <= 1) return;
    int mid = lo + (hi - lo) / 2;
    msort_rec(v, lo, mid);
    msort_rec(v, mid, hi);
    merge_ranges(v, lo, mid, hi);
}

std::vector<int> merge_sort(std::vector<int> v) {
    msort_rec(v, 0, (int)v.size());
    return v;
}

int qsort_partition(std::vector<int> &v, int lo, int hi) {
    int pivot = v[hi], i = lo - 1;
    for (int j = lo; j < hi; j++)
        if (v[j] <= pivot) {
            i++;
            int t = v[i];
            v[i] = v[j];
            v[j] = t;
        }
    int t = v[i + 1];
    v[i + 1] = v[hi];
    v[hi] = t;
    return i + 1;
}

void qsort_rec(std::vector<int> &v, int lo, int hi) {
    if (lo >= hi) return;
    int p = qsort_partition(v, lo, hi);
    qsort_rec(v, lo, p - 1);
    qsort_rec(v, p + 1, hi);
}

std::vector<int> quick_sort(std::vector<int> v) {
    qsort_rec(v, 0, (int)v.size() - 1);
    return v;
}

    for (int i = 0; i < p; i++)

void merge_count(std::vector<int> &v, int lo, int mid, int hi, long long &steps) {
    std::vector<int> t;
    int i = lo, j = mid;
    while (i < mid && j < hi) {
        if (v[i] <= v[j]) {
            t.push_back(v[i]);
            i++;
        } else {
            t.push_back(v[j]);
            j++;
        }
        steps++;
    }
    while (i < mid) {
        t.push_back(v[i]);
        i++;
        steps++;
    }
    while (j < hi) {
        t.push_back(v[j]);
        j++;
        steps++;
    }
    for (int k = 0; k < (int)t.size(); k++) v[lo + k] = t[k];
}

void msort_count_rec(std::vector<int> &v, int lo, int hi, long long &steps) {
    if (hi - lo <= 1) return;
    int mid = lo + (hi - lo) / 2;
    msort_count_rec(v, lo, mid, steps);
    msort_count_rec(v, mid, hi, steps);
    merge_count(v, lo, mid, hi, steps);
}

void merge_sort_count(std::vector<int> &v, long long &steps) {
    steps = 0;
    msort_count_rec(v, 0, (int)v.size(), steps);
}

int lomuto_pick(std::vector<int> &v, int lo, int hi) {
    int pivot = v[hi], i = lo - 1;
    for (int j = lo; j < hi; j++)
        if (v[j] <= pivot) {
            i++;
            int t = v[i];
            v[i] = v[j];
            v[j] = t;
        }
    int t = v[i + 1];
    v[i + 1] = v[hi];
    v[hi] = t;
    return i + 1;

int kth_smallest(std::vector<int> v, int k) {
    int lo = 0, hi = (int)v.size() - 1;
    while (true) {
        int p = lomuto_pick(v, lo, hi);
        if (p == k - 1) return v[p];
        if (p < k - 1) lo = p + 1;
        else hi = p - 1;
    }
}

void three_way_partition(std::vector<int> &v, int pivot) {
    std::vector<int> less, equal, greater;
    for (int x : v) {
        if (x < pivot) less.push_back(x);
        else if (x == pivot) equal.push_back(x);
        else greater.push_back(x);
    }
    int i = 0;
    for (int x : less) v[i++] = x;
    for (int x : equal) v[i++] = x;
    for (int x : greater) v[i++] = x;
}

int hoare_partition(std::vector<int> &v) {
    int n = (int)v.size();
    int pivot = v[0], i = -1, j = n;
    while (true) {
        do { i++; } while (v[i] < pivot);
        do { j--; } while (v[j] > pivot);
        if (i >= j) return j;
        int t = v[i];
        v[i] = v[j];
        v[j] = t;
    }
}

std::vector<int> bottom_up_merge(std::vector<int> v) {
    int n = (int)v.size();
    std::vector<int> t(n);
    for (int width = 1; width < n; width *= 2) {
        for (int lo = 0; lo < n; lo += 2 * width) {
            int mid = lo + width < n ? lo + width : n;
            int hi = lo + 2 * width < n ? lo + 2 * width : n;
            int i = lo, j = mid, k = lo;
            while (i < mid && j < hi)
                t[k++] = v[i] <= v[j] ? v[i++] : v[j++];
            while (i < mid) t[k++] = v[i++];
            while (j < hi) t[k++] = v[j++];
        }
        for (int i = 0; i < n; i++) v[i] = t[i];
    }
    return v;
}
