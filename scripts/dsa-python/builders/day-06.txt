    a = values[:]

    a = values[:]
    for i in range(1, len(a)):
        key = a[i]
        j = i - 1
        while j >= 0 and a[j] > key:

            j -= 1

    return a

def bubble_vs_insertion(values):
    a = values[:]
    n = len(a)
    swaps = 0
    for i in range(n - 1):
        for j in range(n - 1 - i):
            if a[j] > a[j + 1]:
                a[j], a[j + 1] = a[j + 1], a[j]

    b = values[:]

def passes_until_settled(values):
    a = values[:]
    n = len(a)
    passes = 0
    swapped = True
    while swapped:
        swapped = False
        passes += 1
        for j in range(n - 1):
            if a[j] > a[j + 1]:
                a[j], a[j + 1] = a[j + 1], a[j]
                swapped = True
    return passes

def words_by_length_then_alpha(words):
    a = words[:]
    for i in range(1, len(a)):
        cur = a[i]
        j = i - 1
        while j >= 0 and (len(a[j]) > len(cur) or (len(a[j]) == len(cur) and a[j] > cur)):
            a[j + 1] = a[j]
            j -= 1
        a[j + 1] = cur
    return a

def tuples_by_second(pairs):
    a = pairs[:]
    n = len(a)
    for i in range(n):
        m = i
        for j in range(i + 1, n):
            if a[j][1] < a[m][1] or (a[j][1] == a[m][1] and a[j][0] < a[m][0]):
                m = j
        a[i], a[m] = a[m], a[i]
    return a

def bubble_trace(values):
    a = values[:]
    n = len(a)
    states = []
    swapped = True
    while swapped:
        swapped = False
        for j in range(n - 1):
            if a[j] > a[j + 1]:
                a[j], a[j + 1] = a[j + 1], a[j]
                swapped = True
        states.append(a[:])
    return states

def points_by_distance(points):
    a = points[:]
    n = len(a)
    for i in range(n):
        m = i
        for j in range(i + 1, n):
            kj = (a[j][0] * a[j][0] + a[j][1] * a[j][1], a[j][0], a[j][1])
            km = (a[m][0] * a[m][0] + a[m][1] * a[m][1], a[m][0], a[m][1])
            if kj < km:
                m = j
        a[i], a[m] = a[m], a[i]
    return a
