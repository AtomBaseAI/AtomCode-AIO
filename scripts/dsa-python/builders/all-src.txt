        if parts[0] == "push":
            x = int(parts[1])
            stack.append(x)
            if mins and x > mins[-1]:
                mins.append(mins[-1])

### day11[14] Decode the Message Pattern
def decode(s):
    stack = []
    cur = []
    num = 0
    for ch in s:
        if ch.isdigit():
            num = num * 10 + int(ch)
        elif ch == '[':
            stack.append((cur, num))
            cur = []
            num = 0
        elif ch == ']':
            prev, k = stack.pop()
            prev.extend(cur * k)
            cur = prev
        else:
            cur.append(ch)
    return "".join(cur)

### day11[16] Reverse Polish Calculator
def eval_postfix(tokens):
    stack = []
    for tok in tokens:
        if tok in ('+', '-', '*', '/'):
            b = stack.pop()
            a = stack.pop()
            if tok == '+':
                stack.append(a + b)
            elif tok == '-':
                stack.append(a - b)
            elif tok == '*':
                stack.append(a * b)
            else:
                stack.append(int(a / b))
        else:
            stack.append(int(tok))
    return stack[0]

### day12[10] Circular Queue Ticket Desk
def circular_queue_ops(cap, ops):
    buf = [None] * cap
    head = 0
    count = 0
    res = []
    for op in ops:
        parts = op.split()
        if parts[0] == "enq":
            if count < cap:
                buf[(head + count) % cap] = int(parts[1])
                count += 1
        elif parts[0] == "deq":
            if count == 0:
                res.append("EMPTY")
            else:
                res.append(buf[head])
                head = (head + 1) % cap
                count -= 1
        else:
            res.append("EMPTY" if count == 0 else buf[head])
    return res

### day12[13] Round Robin Job Finish Order
def completion_order(bursts, quantum):
    q = deque(range(len(bursts)))
    left = list(bursts)
    done = []
    while q:
        i = q.popleft()
        step = left[i] if left[i] < quantum else quantum
        left[i] -= step
        if left[i] == 0:
            done.append(i)
        else:
            q.append(i)
    return done

### day12[19] Shortest Surge at Least K
def shortest_subarray(nums, target):
    n = len(nums)
    prefix = [0]
    for x in nums:
        prefix.append(prefix[-1] + x)
    dq = deque()
    best = n + 1
    for j in range(n + 1):
        while dq and prefix[j] - prefix[dq[0]] >= target:
            if j - dq[0] < best:
                best = j - dq[0]
            dq.popleft()
        while dq and prefix[dq[-1]] >= prefix[j]:
            dq.pop()
        dq.append(j)
    return best if best <= n else -1

    return root.val + bst_sum(root.left) + bst_sum(root.right)

### day15[7] Pin the Faulty Sensor
def bst_search(root, x):
    node = root
    while node is not None:
        if x == node.val:
            return True
        if x < node.val:
            node = node.left
        else:
            node = node.right
    return False

### day15[10] Kth Smallest Ticket
def kth_smallest(root, k):
    stack = []
    node = root
    while stack or node is not None:
        while node is not None:
            stack.append(node)
            node = node.left
        node = stack.pop()
        k -= 1
        if k == 0:
            return node.val
        node = node.right
    return -1

### day15[12] Priority Desk Operations
def process_ops(ops):
    heap = []
    popped = []
    for op, x in ops:
        if op == "push":
            heapq.heappush(heap, x)
        else:
            if heap:
                popped.append(heapq.heappop(heap))
    return popped

### day15[15] Drain the Priority Heap
def drain_heap(nums):
    heap = list(nums)
    heapq.heapify(heap)
    drained = []
    while heap:
        drained.append(heapq.heappop(heap))
    return drained


### day15[16] Validate the Registry
def is_bst(root):
    def check(node, lo, hi):
        if node is None:
            return True
        if node.val <= lo or node.val >= hi:
            return False
        return check(node.left, lo, node.val) and check(node.right, node.val, hi)
    return check(root, float("-inf"), float("inf"))


### day15[17] Kth Largest via Size-K Heap
def kth_largest(nums, k):
    heap = []
    for x in nums:
        if len(heap) < k:
            heapq.heappush(heap, x)
        elif x > heap[0]:
            heapq.heapreplace(heap, x)
    return heap[0]


### day15[18] Remove a Registry Code
def bst_delete(root, x):
    if root is None:
        return None
    if x < root.val:
        root.left = bst_delete(root.left, x)
    elif x > root.val:
        root.right = bst_delete(root.right, x)
    else:
        if root.left is None:
            return root.right
        if root.right is None:
            return root.left
        successor = root.right
        while successor.left is not None:
            successor = successor.left
        root.val = successor.val
        root.right = bst_delete(root.right, successor.val)
    return root


### day15[19] Codes in the Price Band
def count_range(root, lo, hi):
    if root is None:
        return 0
    if root.val < lo:
        return count_range(root.right, lo, hi)
    if root.val > hi:
        return count_range(root.left, lo, hi)
    return 1 + count_range(root.left, lo, hi) + count_range(root.right, lo, hi)
