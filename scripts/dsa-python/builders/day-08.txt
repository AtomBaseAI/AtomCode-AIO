        return n
    s = 0

def binary_search_recursive(values, x, lo=0, hi=None):
    if hi is None:
        hi = len(values) - 1
    if lo > hi:
        return -1
    mid = (lo + hi) // 2
    if values[mid] == x:
        return mid
    if values[mid] < x:
        return binary_search_recursive(values, x, mid + 1, hi)
    return binary_search_recursive(values, x, lo, mid - 1)

def count_arrangements(letters):
    if not letters:
        return 1
    seen = []
    total = 0
    i = 0
    while i < len(letters):

def fact(m):
    if m <= 1:
        return 1
    return m * fact(m - 1)

def nth_permutation(letters, k):
    if not letters:
        return ""
    block = fact(len(letters) - 1)
    idx = (k - 1) // block
    rest = letters[:idx] + letters[idx + 1:]
    return letters[idx] + nth_permutation(rest, k - idx * block)

def n_queens(n):
    def try_col(row, col, cols, d1, d2):
        if col == n:
            return 0
        total = 0
        if col not in cols and (row + col) not in d1 and (row - col) not in d2:
            if row + 1 == n:
                total += 1
            else:
                total += try_col(row + 1, 0, cols | {col}, d1 | {row + col}, d2 | {row - col})
        return total + try_col(row, col + 1, cols, d1, d2)

    return try_col(0, 0, set(), set(), set())

def count_subsets_neg(values, target, i=0):
    if i == len(values):
        return 1 if target == 0 else 0
    return count_subsets_neg(values, target - values[i], i + 1) + count_subsets_neg(values, target, i + 1)
