def merge_sort(values):
    if len(values) <= 1:
        return values[:]
    mid = len(values) // 2
    left = merge_sort(values[:mid])
    right = merge_sort(values[mid:])
    out = []
    i = j = 0
    while i < len(left) and j < len(right):
        if left[i] <= right[j]:
            out.append(left[i])
            i += 1

            j += 1

        out.append(left[i])
        i += 1
    while j < len(right):
        out.append(right[j])
        j += 1
    return out

    n = len(a)
    pivot = a[n - 1]
    i = -1

def quickselect_kth(values, k):
    a = values[:]
    lo, hi = 0, len(a) - 1
    target = k - 1
    while True:
        pivot = a[hi]
        i = lo - 1
        for j in range(lo, hi):
            if a[j] <= pivot:
                i += 1
                a[i], a[j] = a[j], a[i]
        a[i + 1], a[hi] = a[hi], a[i + 1]
        p = i + 1
        if p == target:
            return a[p]
        if p < target:
            lo = p + 1
        else:
            hi = p - 1

def quicksort_list(values):
    def partition(arr, lo, hi):
        pivot = arr[hi]
        i = lo - 1
        for j in range(lo, hi):
            if arr[j] <= pivot:
                i += 1
                arr[i], arr[j] = arr[j], arr[i]
        arr[i + 1], arr[hi] = arr[hi], arr[i + 1]
        return i + 1

    def quicksort(arr, lo, hi):
        if lo < hi:
            p = partition(arr, lo, hi)
            quicksort(arr, lo, p - 1)
            quicksort(arr, p + 1, hi)

    a = values[:]
    quicksort(a, 0, len(a) - 1)
    return a

def inversion_count(values):
    def sort_count(arr):
        if len(arr) <= 1:
            return arr[:], 0
        mid = len(arr) // 2
        left, li = sort_count(arr[:mid])
        right, ri = sort_count(arr[mid:])
        merged = []
        inv = li + ri
        i = j = 0
        while i < len(left) and j < len(right):
            if left[i] <= right[j]:
                merged.append(left[i])
                i += 1
            else:
                merged.append(right[j])
                j += 1
                inv += len(left) - i
        while i < len(left):
            merged.append(left[i])
            i += 1
        while j < len(right):
            merged.append(right[j])
            j += 1
        return merged, inv

    return sort_count(values)[1]

def median_two(a, b):
    need = (len(a) + len(b)) // 2
    i = j = 0
    last = 0
    while i + j <= need:
        if i < len(a) and (j >= len(b) or a[i] <= b[j]):
            last = a[i]
            i += 1
        else:
            last = b[j]
            j += 1
    return last

def minimum_swaps(values):
    a = values[:]
    n = len(a)
    ranks = {}
    for v in a:
        r = 0
        for w in a:
            if w < v:
                r += 1
        ranks[v] = r
    swaps = 0
    i = 0
    while i < n:
        j = ranks[a[i]]
        if j != i:
            a[i], a[j] = a[j], a[i]
            swaps += 1
        else:
            i += 1
    return swaps

def leaderboard(pairs):
    a = pairs[:]
    n = len(a)
    for i in range(n):
        m = i
        for j in range(i + 1, n):
            if a[j][1] > a[m][1] or (a[j][1] == a[m][1] and a[j][0] < a[m][0]):
                m = j
        a[i], a[m] = a[m], a[i]
    return [p[0] for p in a]
