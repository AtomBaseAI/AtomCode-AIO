    tail = None
    for v in values:
        node = Node(v)

        tail = node
    return head

def neighbours(head, i):
    node = head
    k = 0
    while k < i:
        node = node.next
        k += 1
    left = node.prev.value if node.prev is not None else -1
    right = node.next.value if node.next is not None else -1
    return [left, node.value, right]

def josephus_survivor(head, k):
    prev = head
    while prev.next is not head:
        prev = prev.next
    while prev.next is not prev:
        i = 1
        while i < k:
            prev = prev.next
            i += 1
        prev.next = prev.next.next
    return prev.value

def reverse_dll(head):
    node = head
    new_head = head
    while node is not None:
        node.prev, node.next = node.next, node.prev
        new_head = node
        node = node.prev
