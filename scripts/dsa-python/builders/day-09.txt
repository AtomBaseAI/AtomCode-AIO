def delete_first(head, x):
    if head is None:
        return None
    if head.value == x:
        return head.next
    prev = head
    while prev.next is not None:
        if prev.next.value == x:
            prev.next = prev.next.next
            break
        prev = prev.next
    return head

    node = Node(x)
    if head is None or x < head.value:

    node = head

def delete_all(head, x):
    while head is not None and head.value == x:
        head = head.next
    if head is None:
        return None
    prev = head
    while prev.next is not None:
        if prev.next.value == x:
            prev.next = prev.next.next
        else:
            prev = prev.next
    return head

def merge_chains(head1, head2):
    dummy = Node(0)
    tail = dummy
    a, b = head1, head2
    while a is not None and b is not None:
        if a.value <= b.value:
            tail.next = a
            a = a.next
        else:
            tail.next = b
            b = b.next
        tail = tail.next
    if a is not None:
        tail.next = a
    else:
        tail.next = b
    return dummy.next

def cycle_start(head):
    slow = head
    fast = head
    while fast is not None and fast.next is not None:
        slow = slow.next
        fast = fast.next.next
        if slow is fast:
            finder = head
            while finder is not slow:
                finder = finder.next
                slow = slow.next
            return finder.value
    return -1

def reverse_groups(head, k):
    node = head
    count = 0
    while node is not None:
        count += 1
        node = node.next
    new_head = None
    prev_tail = None
    curr = head
    while count >= k:
        group_head = curr
        prev = None
        i = 0
        while i < k:
            nxt = curr.next
            curr.next = prev
            prev = curr
            curr = nxt
            i += 1
        if new_head is None:
            new_head = prev
        else:
            prev_tail.next = prev
        prev_tail = group_head
        count -= k
    if prev_tail is not None:
        prev_tail.next = curr
    if new_head is None:
        return head
    return new_head
