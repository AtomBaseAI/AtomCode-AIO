void sift_down(int a[], int n, int i) {
    while (1) {
        int smallest = i;
        int l = 2 * i + 1;
        int r = 2 * i + 2;
        if (l < n && a[l] < a[smallest]) smallest = l;
        if (r < n && a[r] < a[smallest]) smallest = r;
        if (smallest == i) break;
        int t = a[i];
        a[i] = a[smallest];
        a[smallest] = t;
        i = smallest;
    }
}

void build_min_heap(int a[], int n) {
    for (int i = n / 2 - 1; i >= 0; i--) sift_down(a, n, i);
}

void heap_push(int heap[], int *size, int v) {
    int i = (*size)++;
    heap[i] = v;
    while (i > 0) {
        int p = (i - 1) / 2;
        if (heap[p] <= heap[i]) break;
        int t = heap[p];
        heap[p] = heap[i];
        heap[i] = t;
        i = p;
    }
}

int heap_pop(int heap[], int *size) {
    int top = heap[0];
    (*size)--;
    heap[0] = heap[*size];
    int i = 0;
    while (1) {
        int l = 2 * i + 1;
        int r = 2 * i + 2;
        int smallest = i;
        if (l < *size && heap[l] < heap[smallest]) smallest = l;
        if (r < *size && heap[r] < heap[smallest]) smallest = r;
        if (smallest == i) break;
        int t = heap[i];
        heap[i] = heap[smallest];
        heap[smallest] = t;
        i = smallest;
    }
    return top;
}

int bst_closest(BNode *root, int target) {
    int best = root->key;
    BNode *cur = root;
    while (cur != NULL) {
        if (abs(cur->key - target) < abs(best - target)) best = cur->key;
        if (target < cur->key) cur = cur->left;
        else if (target > cur->key) cur = cur->right;
        else break;
    }
    return best;
}
