Output Format:
One line with YES or NO.

Constraints:

TC s1 IN<listen silent> EXP<YES>
TC s2 IN<triangle integral> EXP<YES>
TC t1 IN<hello world> EXP<NO>

SRC:
int is_anagram(const char *a, const char *b) {
    int freq[26] = {0};
    int la = 0, lb = 0;
    while (a[la]) la++;
    while (b[lb]) lb++;
    if (la != lb) return 0;
    for (int i = 0; i < la; i++) freq[a[i] - 'a']++;
    for (int i = 0; i < lb; i++) {
        freq[b[i] - 'a']--;
        if (freq[b[i] - 'a'] < 0) return 0;
    }
    return 1;
}

int subarray_sum_count(const int *arr, int n, int k) {
    int freq[20001];
    for (int i = 0; i <= 20000; i++) freq[i] = 0;
    int pre = 0, count = 0;
    freq[10000] = 1;
    for (int i = 0; i < n; i++) {
        pre += arr[i];
        int need = pre - k;
        if (need >= -10000 && need <= 10000) count += freq[need + 10000];
        freq[pre + 10000]++;
    }
    return count;
}
